// Módulo Identificação de Equipamentos - injetado no iframe FireCheck (executa dentro do iframe, App é global)

const EQUIP_TYPES = {
    H: 'Hidrante', PC: 'Porta Corta Fogo', EX: 'Extintor', MG: 'Mangueira',
    DF: 'Detector de Fumaça', AM: 'Acionador Manual', SAV: 'Sinalizador', PCH: 'Prancha'
};

const EQUIP_STATUS_BADGE = {
    'Conforme': 'bg-emerald-100 text-emerald-700',
    'Não Conforme': 'bg-red-100 text-red-700',
    'Em Manutenção': 'bg-amber-100 text-amber-700'
};

const EQUIP_PER_PAGE = 10;

App.prototype.equipCanManage = function() {
    return this.currentUser && ['CILO', 'DIREX'].includes(this.currentUser.nivel);
};

App.prototype.renderEquipamentos = function() {
    if (this.relatorioDetalhe) return this.renderRelatorioDetalhe();
    this.equipTab = this.equipTab || 'cadastro';
    this.equipCodigos = this.equipCodigos || [];
    this.equipLoading = this.equipLoading || false;
    this.equipEditing = this.equipEditing || null;
    this.equipRelatorios = this.equipRelatorios || [];
    this.equipRelatoriosLoading = this.equipRelatoriosLoading || false;
    this.equipPage = this.equipPage || 1;
    this.equipSearchTerm = this.equipSearchTerm || '';
    this.equipRelSearchTerm = this.equipRelSearchTerm || '';

    const tab = this.equipTab;
    const tabBtn = (id, label, icon) => `
        <button onclick="app.equipSetTab('${id}')" class="flex items-center gap-2 px-5 py-3 text-xs font-bold rounded-t-xl transition-all ${tab === id ? 'bg-white text-slate-900 border-t-2 border-fire-600' : 'text-slate-500 hover:text-slate-700 bg-slate-100'}">
            <span>${icon}</span> ${label}
        </button>`;

    return `<div class="space-y-0">
        <div class="flex border-b border-slate-200 bg-slate-50 rounded-t-xl">
            ${tabBtn('cadastro', 'Cadastro / Edição de Códigos', '📋')}
            ${tabBtn('consulta', 'Consulta de Relatórios Emitidos', '📊')}
        </div>
        <div class="bg-white rounded-b-xl shadow-sm border border-slate-200 border-t-0 p-6">
            ${tab === 'cadastro' ? this.renderEquipCadastro() : this.renderEquipConsulta()}
        </div>
    </div>`;
};

App.prototype.equipSetTab = function(tab) {
    this.equipTab = tab;
    if (tab === 'consulta' && this.equipRelatorios.length === 0 && !this.equipRelatoriosLoading) {
        this.equipBuscarRelatorios();
    }
    this.render();
};

App.prototype.equipRenderPagination = function(totalPages) {
    let btns = '';
    for (let i = 1; i <= totalPages; i++) {
        btns += `<button onclick="app.equipSetPage(${i})" class="px-3 py-1.5 text-xs font-bold rounded-lg transition-colors ${i === this.equipPage ? 'bg-slate-900 text-white' : 'text-slate-600 hover:bg-slate-100 border border-slate-200'}">${i}</button>`;
    }
    return `<div class="flex justify-center items-center gap-2 mt-4 no-print">${btns}</div>`;
};

App.prototype.equipSetPage = function(page) {
    this.equipPage = page;
    this.render();
};

App.prototype.equipCarregarLista = function() {
    this.equipLoading = true;
    window.parent.postMessage({ type: 'firecheck-equipamento-listar' }, '*');
};

App.prototype.equipGetFiltered = function() {
    const term = (this.equipSearchTerm || '').toLowerCase().trim();
    if (!term) return this.equipCodigos || [];
    return (this.equipCodigos || []).filter(e =>
        (e.codigo || '').toLowerCase().includes(term) ||
        (e.tipo || '').toLowerCase().includes(term) ||
        (EQUIP_TYPES[e.tipo] || '').toLowerCase().includes(term) ||
        (e.bloco || '').toLowerCase().includes(term) ||
        (e.andar || '').toLowerCase().includes(term) ||
        (e.descricao || '').toLowerCase().includes(term)
    );
};

App.prototype.equipFiltrar = function(val) {
    this.equipSearchTerm = val;
    this.equipPage = 1;
    this.render();
};

App.prototype.equipAtualizarPreview = function() {
    const tipo = document.getElementById('eq-tipo')?.value || '';
    const bloco = document.getElementById('eq-bloco')?.value || '';
    const andar = document.getElementById('eq-andar')?.value || '';
    const num = document.getElementById('eq-numero')?.value || '';
    const andarStr = andar === 'T' ? 'T' : (andar ? String(andar).padStart(2, '0') : '');
    const numStr = num ? String(num).padStart(2, '0') : '';
    const preview = tipo && bloco && andarStr && numStr ? `${tipo}-32-${bloco}-${andarStr}-${numStr}` : '--';
    const el = document.getElementById('eq-preview');
    if (el) el.textContent = preview;
};

App.prototype.equipSolicitarSugestao = function() {
    this.equipAtualizarPreview();
    const tipo = document.getElementById('eq-tipo')?.value;
    const bloco = document.getElementById('eq-bloco')?.value;
    const andar = document.getElementById('eq-andar')?.value;
    if (!tipo || !bloco || !andar) return;
    if (this.equipEditing) return; // não sugerir durante edição
    window.parent.postMessage({ type: 'firecheck-equipamento-sugerir', payload: { action: 'sugerir', tipo, bloco, andar: String(andar) } }, '*');
};

App.prototype.equipSalvar = function() {
    const tipo = document.getElementById('eq-tipo')?.value;
    const bloco = document.getElementById('eq-bloco')?.value;
    const andar = document.getElementById('eq-andar')?.value?.trim();
    const numero = document.getElementById('eq-numero')?.value;
    const descricao = document.getElementById('eq-descricao')?.value?.trim() || '';
    const dataInspecao = document.getElementById('eq-data-inspecao')?.value || '';
    const status = document.getElementById('eq-status')?.value || 'Conforme';
    const obs = document.getElementById('eq-obs')?.value?.trim() || '';

    if (!tipo || !bloco || !andar || !numero) {
        return this.showModal('Campos Obrigatórios', 'Preencha Tipo, Bloco, Andar e Número do Equipamento.');
    }
    if (andar !== 'T' && isNaN(Number(andar))) {
        return this.showModal('Andar Inválido', 'O andar deve ser um número ou "T" para térreo.');
    }

    const payload = {
        action: this.equipEditing ? 'editar' : 'criar',
        tipo, bloco, andar: String(andar), numero_equipamento: Number(numero),
        descricao, data_ultima_inspecao: dataInspecao, status, observacoes: obs,
        responsavel: this.currentUser ? this.currentUser.nome : ''
    };
    if (this.equipEditing) payload.id = this.equipEditing;

    this.showToast('Salvando código...', 'success');
    window.parent.postMessage({ type: 'firecheck-equipamento-salvar', payload }, '*');
};

App.prototype.equipEditar = function(id) {
    this.equipEditing = id;
    this.equipPage = 1;
    this.render();
    setTimeout(() => this.equipAtualizarPreview(), 50);
};

App.prototype.equipCancelarEdicao = function() {
    this.equipEditing = null;
    this.render();
};

App.prototype.equipExcluir = function(id) {
    this.showModal('Excluir Código', 'Tem certeza que deseja excluir este código de equipamento? Esta ação não pode ser desfeita.', () => {
        window.parent.postMessage({ type: 'firecheck-equipamento-excluir', payload: { action: 'excluir', id } }, '*');
    });
};

App.prototype.equipBuscarRelatorios = function() {
    const term = document.getElementById('eq-rel-search')?.value?.trim() || '';
    this.equipRelSearchTerm = term;
    this.equipRelatoriosLoading = true;
    this.render();
    window.parent.postMessage({
        type: 'firecheck-equipamento-consultar-relatorios',
        filtros: { equipamento: term, limit: 200 }
    }, '*');
};

App.prototype.equipExportarCSV = function() {
    const filtered = this.equipGetFiltered();
    if (filtered.length === 0) return this.showToast('Nenhum código para exportar', 'error');
    const headers = ['Código', 'Tipo', 'Equipamento', 'Prédio', 'Bloco', 'Andar', 'Nº Equip', 'Status', 'Descrição', 'Data Última Inspeção', 'Responsável', 'Observações'];
    const rows = filtered.map(e => [
        e.codigo, e.tipo, EQUIP_TYPES[e.tipo] || '', e.predio, e.bloco, e.andar,
        e.numero_equipamento, e.status, e.descricao, e.data_ultima_inspecao, e.responsavel_cadastro, e.observacoes
    ]);
    const csv = [headers, ...rows].map(r => r.map(c => `"${String(c || '').replace(/"/g, '""')}"`).join(',')).join('\n');
    const blob = new Blob(['\ufeff' + csv], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = 'equipamentos_firecheck.csv';
    a.click();
    URL.revokeObjectURL(url);
    this.showToast('✓ Exportação CSV gerada com ' + filtered.length + ' registro(s)', 'success');
};

// Listener para resultados das operações de equipamento
window.addEventListener('message', function(event) {
    if (!event.data) return;
    const app = window.app;
    if (!app) return;

    if (event.data.type === 'firecheck-equipamento-listar-result') {
        app.equipLoading = false;
        if (event.data.success) {
            app.equipCodigos = event.data.codigos || [];
        } else {
            app.equipCodigos = [];
            app.showToast('✕ ' + (event.data.message || 'Erro ao listar códigos'), 'error');
        }
        if (app.currentModule === 'equipamentos') app.render();
    }
    else if (event.data.type === 'firecheck-equipamento-sugerir-result') {
        if (event.data.success) {
            const numInput = document.getElementById('eq-numero');
            if (numInput && !app.equipEditing) {
                numInput.value = event.data.proximo_numero;
            }
            app.equipAtualizarPreview();
        }
    }
    else if (event.data.type === 'firecheck-equipamento-salvar-result') {
        if (event.data.success) {
            app.showToast('✓ ' + event.data.message, 'success');
            app.equipEditing = null;
            app.equipamentosCache = {}; // Invalida cache dos seletores em silo
            app.equipCarregarLista();
            app.render();
        } else {
            const msg = event.data.message || '';
            if (msg.includes('duplicidade')) {
                app.showModal('Duplicidade de Código', msg);
            } else {
                app.showToast('✕ ' + msg, 'error');
            }
        }
    }
    else if (event.data.type === 'firecheck-equipamento-excluir-result') {
        if (event.data.success) {
            app.showToast('✓ ' + event.data.message, 'success');
            app.equipamentosCache = {}; // Invalida cache dos seletores em silo
            app.equipCarregarLista();
        } else {
            app.showToast('✕ ' + event.data.message, 'error');
        }
    }
    else if (event.data.type === 'firecheck-equipamento-consultar-relatorios-result') {
        app.equipRelatoriosLoading = false;
        if (event.data.success) {
            app.equipRelatorios = event.data.relatorios || [];
            if (app.equipRelatorios.length === 0) {
                app.showToast('Nenhum relatório encontrado para este código de equipamento', 'success');
            } else {
                app.showToast('✓ ' + (event.data.total || 0) + ' relatório(s) encontrado(s)', 'success');
            }
        } else {
            app.equipRelatorios = [];
            app.showToast('✕ ' + (event.data.message || 'Erro na consulta'), 'error');
        }
        if (app.currentModule === 'equipamentos') app.render();
    }
});
