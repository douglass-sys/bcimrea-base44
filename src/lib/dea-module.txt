const DEA_FREQ_LABEL = {
    'diaria': 'Diária', 'semanal': 'Semanal', 'mensal': 'Mensal',
    'trimestral': 'Trimestral', 'anual': 'Anual'
};

const DEA_FREQ_DAYS = {
    'diaria': 1, 'semanal': 7, 'mensal': 30, 'trimestral': 90, 'anual': 365
};

const DEA_VERIFICACAO_ITENS = [

const DEA_MANUT_STATUS_BADGE = {
    'Planejada': 'bg-blue-100 text-blue-700',
    'Em Execução': 'bg-amber-100 text-amber-700',
    'Concluída': 'bg-emerald-100 text-emerald-700',

App.prototype.deaCanManage = function() {
    return this.currentUser && ['CILO', 'DIREX', 'BOMBEIRO'].includes(this.currentUser.nivel);
};

App.prototype.deaCanEdit = function() {
    return this.currentUser && ['CILO', 'DIREX'].includes(this.currentUser.nivel);
};

// ── RENDER PRINCIPAL ──────────────────────────────────────────
App.prototype.renderDEA = function() {
    this.deaTab = this.deaTab || 'painel';
    this.deaList = this.deaList || [];
    this.deaVerificacoes = this.deaVerificacoes || [];
    this.deaManutencoes = this.deaManutencoes || [];
    this.deaLoading = this.deaLoading || false;
    this.deaEditing = this.deaEditing || null;
    this.deaSearchTerm = this.deaSearchTerm || '';
    this.deaVerFilterDea = this.deaVerFilterDea || '';
    this.deaManFilterDea = this.deaManFilterDea || '';

    const tab = this.deaTab;
    const tabBtn = (id, label, icon) => `
        <button onclick="app.deaSetTab('${id}')" class="flex items-center gap-2 px-5 py-3 text-xs font-bold rounded-t-xl transition-all ${tab === id ? 'bg-white text-slate-900 border-t-2 border-fire-600' : 'text-slate-500 hover:text-slate-700 bg-slate-100'}">
            <span>${icon}</span> ${label}
        </button>`;

    return `<div class="space-y-0">
        <div class="flex border-b border-slate-200 bg-slate-50 rounded-t-xl overflow-x-auto">
            ${tabBtn('painel', 'Painel Inicial', '📊')}

            ${tabBtn('verificacoes', 'Testes e Verificações', '✅')}

            ${tab === 'painel' ? this.deaRenderPainel() : ''}
            ${tab === 'equipamentos' ? this.deaRenderEquipamentos() : ''}
            ${tab === 'verificacoes' ? this.deaRenderVerificacoes() : ''}
            ${tab === 'manutencao' ? this.deaRenderManutencao() : ''}
            ${tab === 'relatorios' ? this.deaRenderRelatorios() : ''}
        </div>

};

App.prototype.deaSetTab = function(tab) {
    this.deaTab = tab;
    this.deaEditing = null;
    if (tab === 'painel') this.deaCarregarTudo();
    if (tab === 'equipamentos') this.deaCarregar();
    if (tab === 'verificacoes') { this.deaCarregar(); this.deaCarregarVerificacoes(); }
    if (tab === 'manutencao') { this.deaCarregar(); this.deaCarregarManutencoes(); }
    this.render();
};

// ── PAINEL INICIAL ────────────────────────────────────────────
App.prototype.deaDaysLeft = function(dateStr) {
    if (!dateStr) return null;
    const d = new Date(dateStr);
    if (isNaN(d)) return null;
    return Math.ceil((d - new Date()) / 86400000);
};

App.prototype.deaRenderPainel = function() {
    const deas = (this.deaList || []).filter(d => d.ativo !== false);
    const total = deas.length;
    const operando = deas.filter(d => d.status === 'Operando').length;
    const manutencao = deas.filter(d => d.status === 'Em Manutenção').length;
    const foraUso = deas.filter(d => d.status === 'Fora de Uso').length;
    const pendente = deas.filter(d => d.status === 'Pendente Verificação').length;

    // Alertas de validade
    let batVencidas = 0, batProximas = 0;
    let eletVencidos = 0, eletProximos = 0;
    deas.forEach(d => {
        const batDays = this.deaDaysLeft(d.validade_bateria);
        if (batDays !== null) { if (batDays < 0) batVencidas++; else if (batDays <= 30) batProximas++; }
        const eletDays = this.deaDaysLeft(d.validade_eletrodos);
        if (eletDays !== null) { if (eletDays < 0) eletVencidos++; else if (eletDays <= 30) eletProximos++; }
    });

    // Verificações atrasadas
    const verifs = this.deaVerificacoes || [];
    let verAtrasadas = 0;
    deas.forEach(d => {
        const freq = d.frequencia_verificacao || 'mensal';
        const days = DEA_FREQ_DAYS[freq] || 30;
        const deaVerifs = verifs.filter(v => v.dea_id === d.id).sort((a,b) => new Date(b.data_verificacao) - new Date(a.data_verificacao));
        if (deaVerifs.length === 0) { verAtrasadas++; return; }
        const lastVer = new Date(deaVerifs[0].data_verificacao);
        const dueDate = new Date(lastVer.getTime() + days * 86400000);
        if (dueDate < new Date()) verAtrasadas++;
    });

    // Manutenções atrasadas/pendentes
    const manuts = this.deaManutencoes || [];
    const manAtrasadas = manuts.filter(m => m.status !== 'Concluída' && m.data_prevista && new Date(m.data_prevista) < new Date()).length;
    const manPendentes = manuts.filter(m => m.status === 'Planejada' || m.status === 'Em Execução').length;

    const totalAlertas = batVencidas + batProximas + eletVencidos + eletProximos + verAtrasadas + manAtrasadas;

    if (this.deaLoading) {
        return `<div class="p-12 text-center"><div class="inline-block w-8 h-8 border-4 border-slate-200 border-t-slate-800 rounded-full animate-spin"></div><p class="text-xs text-slate-500 mt-3">Carregando dados...</p></div>`;
    }

    const statCard = (label, value, color, icon) => `
        <div class="bg-white p-5 rounded-2xl shadow-sm border border-slate-200">
            <div class="flex items-center justify-between mb-2">
                <span class="text-[10px] font-bold text-slate-400 uppercase tracking-wider">${label}</span>
                <span class="text-lg">${icon}</span>
            </div>
            <div class="text-3xl font-extrabold ${color}">${value}</div>
        </div>`;

    const alertCard = (count, label, color, icon) => count > 0 ? `
        <div class="flex-1 ${color} border rounded-xl p-4 flex items-center gap-3">
            <span class="text-2xl">${icon}</span>
            <div><div class="text-lg font-bold">${count}</div><div class="text-[10px] font-semibold uppercase tracking-wide">${label}</div></div>
        </div>` : '';

    return `
    <div class="space-y-6">
        <div><h3 class="font-bold text-slate-800 text-lg mb-1">Painel Geral — Desfibriladores (DEA)</h3>
        <p class="text-xs text-slate-500">Visão geral do parque de DEA, verificações periódicas e manutenção preventiva.</p></div>

        <div class="grid grid-cols-2 md:grid-cols-5 gap-4">
            ${statCard('Total de DEAs', total, 'text-slate-800', '📋')}
            ${statCard('Operando', operando, 'text-emerald-600', '✅')}
            ${statCard('Em Manutenção', manutencao, 'text-amber-600', '🔧')}
            ${statCard('Fora de Uso', foraUso, 'text-red-600', '🚫')}
            ${statCard('Pendente Verificação', pendente, 'text-slate-500', '⏳')}
        </div>

        ${totalAlertas > 0 ? `
        <div class="bg-red-50 border border-red-200 rounded-xl p-4">
            <h4 class="text-xs font-bold text-red-700 uppercase mb-3 flex items-center gap-2">⚠ Alertas Ativos (${totalAlertas})</h4>
            <div class="flex flex-wrap gap-3">
                ${alertCard(batVencidas, 'Baterias Vencidas', 'bg-red-100 text-red-700', '🔋')}
                ${alertCard(batProximas, 'Baterias a Vencer (30d)', 'bg-amber-100 text-amber-700', '🔋')}
                ${alertCard(eletVencidos, 'Eletrodos Vencidos', 'bg-red-100 text-red-700', '⚡')}
                ${alertCard(eletProximos, 'Eletrodos a Vencer (30d)', 'bg-amber-100 text-amber-700', '⚡')}
                ${alertCard(verAtrasadas, 'Verificações Atrasadas', 'bg-red-100 text-red-700', '✅')}
                ${alertCard(manAtrasadas, 'Manutenções Atrasadas', 'bg-red-100 text-red-700', '🔧')}
            </div>
        </div>` : `
        <div class="bg-emerald-50 border border-emerald-200 rounded-xl p-4 flex items-center gap-3">
            <span class="text-2xl">✅</span>
            <div><div class="text-sm font-bold text-emerald-700">Sem alertas ativos</div><div class="text-[10px] text-emerald-600">Todos os DEAs estão em conformidade com o cronograma.</div></div>
        </div>`}

        <div class="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div class="bg-white border border-slate-200 rounded-xl p-4">
                <h4 class="text-xs font-bold text-slate-600 uppercase mb-2">Manutenções Pendentes</h4>
                <div class="text-2xl font-bold text-slate-800">${manPendentes}</div>
                <button onclick="app.deaSetTab('manutencao')" class="mt-2 text-[10px] font-bold text-fire-700 hover:underline">Ver detalhes →</button>
            </div>
            <div class="bg-white border border-slate-200 rounded-xl p-4">
                <h4 class="text-xs font-bold text-slate-600 uppercase mb-2">Verificações Registradas</h4>
                <div class="text-2xl font-bold text-slate-800">${verifs.length}</div>
                <button onclick="app.deaSetTab('verificacoes')" class="mt-2 text-[10px] font-bold text-fire-700 hover:underline">Ver histórico →</button>
            </div>
        </div>
    </div>`;
};

// ── CRUD E DATA LOADING ───────────────────────────────────────
App.prototype.deaGetFiltered = function() {
    const term = (this.deaSearchTerm || '').toLowerCase().trim();
    const all = (this.deaList || []).filter(d => d.ativo !== false);
    if (!term) return all;
    return all.filter(d =>
        (d.patrimonio || '').toLowerCase().includes(term) ||
        (d.localizacao || '').toLowerCase().includes(term) ||
        (d.modelo || '').toLowerCase().includes(term) ||
        (d.numero_serie || '').toLowerCase().includes(term)
    );
};

App.prototype.deaSalvar = function() {
    const patrimonio = document.getElementById('dea-patrimonio')?.value?.trim();
    const localizacao = document.getElementById('dea-localizacao')?.value?.trim();
    if (!patrimonio || !localizacao) return this.showModal('Campos Obrigatórios', 'Preencha Patrimônio e Localização.');

    const payload = {
        patrimonio,
        localizacao,
        modelo: document.getElementById('dea-modelo')?.value?.trim() || '',
        numero_serie: document.getElementById('dea-numero-serie')?.value?.trim() || '',
        data_instalacao: document.getElementById('dea-data-instalacao')?.value || '',
        validade_bateria: document.getElementById('dea-validade-bateria')?.value || '',
        validade_eletrodos: document.getElementById('dea-validade-eletrodos')?.value || '',
        status: document.getElementById('dea-status')?.value || 'Pendente Verificação',
        frequencia_verificacao: document.getElementById('dea-frequencia')?.value || 'mensal',
        observacoes: document.getElementById('dea-obs')?.value?.trim() || '',
        ativo: true,
        operador: this.currentUser?.nome || '',
        operador_id: this.currentUser?.matricula || '',
        operador_perfil: this.currentUser?.nivel || ''
    };
    if (this.deaEditing) payload.id = this.deaEditing;

    this.showToast('Salvando DEA...', 'success');
    window.parent.postMessage({ type: 'firecheck-dea-salvar', payload }, '*');
};

App.prototype.deaExportCSV = function(filename, headers, rows) {
    const csv = [headers, ...rows].map(r => r.map(c => `"${String(c || '').replace(/"/g, '""')}"`).join(',')).join('\n');
    const blob = new Blob(['\ufeff' + csv], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url; a.download = filename; a.click();
    URL.revokeObjectURL(url);
    this.showToast('✓ Exportação CSV gerada', 'success');
};

App.prototype.deaExportarDEAsCSV = function() {
    const deas = this.deaGetFiltered();
    if (deas.length === 0) return this.showToast('Nenhum DEA para exportar', 'error');
    const headers = ['Patrimônio', 'Localização', 'Modelo', 'Nº Série', 'Instalação', 'Validade Bateria', 'Validade Eletrodos', 'Status', 'Frequência', 'Observações'];
    const rows = deas.map(d => [d.patrimonio, d.localizacao, d.modelo, d.numero_serie, d.data_instalacao, d.validade_bateria, d.validade_eletrodos, d.status, DEA_FREQ_LABEL[d.frequencia_verificacao] || '', d.observacoes]);
    this.deaExportCSV('dea_equipamentos.csv', headers, rows);
};

App.prototype.deaExportarVerificacoesCSV = function() {
    const verifs = [...(this.deaVerificacoes || [])].sort((a,b) => new Date(b.data_verificacao) - new Date(a.data_verificacao));

    const headers = ['DEA', 'Data', 'Responsável', 'Integridade', 'Indicadores', 'Bateria', 'Eletrodos', 'Sinal Sonoro', 'Auto-teste', 'Resultado', 'Observações'];
    const rows = verifs.map(v => {

};

App.prototype.deaExportarManutencoesCSV = function() {

App.prototype.deaExportarAlertasCSV = function() {
    const deas = (this.deaList || []).filter(d => d.ativo !== false);
    const verifs = this.deaVerificacoes || [];
    const manuts = this.deaManutencoes || [];
    const rows = [];
    deas.forEach(d => {
        const batDays = this.deaDaysLeft(d.validade_bateria);
        const eletDays = this.deaDaysLeft(d.validade_eletrodos);
        if (batDays !== null && batDays < 0) rows.push([d.patrimonio, 'Bateria Vencida', d.validade_bateria, `${Math.abs(batDays)} dias`, 'Substituir imediatamente']);
        if (batDays !== null && batDays >= 0 && batDays <= 30) rows.push([d.patrimonio, 'Bateria a Vencer', d.validade_bateria, `${batDays} dias`, 'Programar substituição']);
        if (eletDays !== null && eletDays < 0) rows.push([d.patrimonio, 'Eletrodos Vencidos', d.validade_eletrodos, `${Math.abs(eletDays)} dias`, 'Substituir imediatamente']);
        if (eletDays !== null && eletDays >= 0 && eletDays <= 30) rows.push([d.patrimonio, 'Eletrodos a Vencer', d.validade_eletrodos, `${eletDays} dias`, 'Programar substituição']);
        const freq = d.frequencia_verificacao || 'mensal';
        const days = DEA_FREQ_DAYS[freq] || 30;
        const deaVerifs = verifs.filter(v => v.dea_id === d.id).sort((a,b) => new Date(b.data_verificacao) - new Date(a.data_verificacao));
        if (deaVerifs.length === 0) { rows.push([d.patrimonio, 'Verificação Pendente', '—', 'Nunca verificada', 'Realizar verificação inicial']); }
        else {
            const lastVer = new Date(deaVerifs[0].data_verificacao);
            const dueDate = new Date(lastVer.getTime() + days * 86400000);
            if (dueDate < new Date()) rows.push([d.patrimonio, 'Verificação Atrasada', deaVerifs[0].data_verificacao, `${Math.ceil((new Date() - dueDate) / 86400000)} dias de atraso`, 'Realizar verificação']);
        }
    });
    manuts.forEach(m => {
        if (m.status !== 'Concluída' && m.data_prevista && new Date(m.data_prevista) < new Date())
            rows.push([m.dea_patrimonio, 'Manutenção Atrasada', m.data_prevista, m.status, 'Executar ou replanejar']);
    });
    if (rows.length === 0) return this.showToast('Nenhum alerta para exportar', 'success');
    this.deaExportCSV('dea_alertas.csv', ['DEA', 'Tipo de Alerta', 'Data Referência', 'Detalhe', 'Ação Recomendada'], rows);
};

// ── MESSAGE LISTENER ─────────────────────────────────────────
window.addEventListener('message', function(event) {
    if (!event.data) return;
    const app = window.app;
    if (!app) return;

    if (event.data.type === 'firecheck-dea-listar-result') {
        app.deaLoading = false;
        if (event.data.success) app.deaList = event.data.deas || [];
        if (app.currentModule === 'dea') app.render();
    }
    else if (event.data.type === 'firecheck-dea-salvar-result') {
        if (event.data.success) {
            app.showToast('✓ ' + event.data.message, 'success');
            app.deaEditing = null;
            app.deaCarregar();
        } else {
            app.showToast('✕ ' + (event.data.message || 'Erro ao salvar'), 'error');
        }
    }
    else if (event.data.type === 'firecheck-dea-arquivar-result') {
        if (event.data.success) {
            app.showToast('✓ ' + event.data.message, 'success');
            app.deaCarregar();
        } else {
            app.showToast('✕ ' + (event.data.message || 'Erro ao arquivar'), 'error');
        }
    }
    else if (event.data.type === 'firecheck-dea-verificacao-listar-result') {
        if (event.data.success) app.deaVerificacoes = event.data.verificacoes || [];
        if (app.currentModule === 'dea') app.render();
    }
    else if (event.data.type === 'firecheck-dea-verificacao-salvar-result') {
        if (event.data.success) {
            app.showToast('✓ ' + event.data.message, 'success');
            app.deaCarregarVerificacoes();
        } else {
            app.showToast('✕ ' + (event.data.message || 'Erro ao registrar'), 'error');
        }
    }
    else if (event.data.type === 'firecheck-dea-manutencao-listar-result') {
        if (event.data.success) app.deaManutencoes = event.data.manutencoes || [];
        if (app.currentModule === 'dea') app.render();
    }
    else if (event.data.type === 'firecheck-dea-manutencao-salvar-result') {
        if (event.data.success) {
            app.showToast('✓ ' + event.data.message, 'success');
            app.deaCarregarManutencoes();
        } else {
            app.showToast('✕ ' + (event.data.message || 'Erro ao criar'), 'error');
        }
    }
    else if (event.data.type === 'firecheck-dea-manutencao-concluir-result') {
        if (event.data.success) {
            app.showToast('✓ ' + event.data.message, 'success');
            app.deaCarregarManutencoes();
        } else {
            app.showToast('✕ ' + (event.data.message || 'Erro ao concluir'), 'error');
        }
    }
});
